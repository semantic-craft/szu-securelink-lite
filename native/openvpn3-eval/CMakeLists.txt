cmake_minimum_required(VERSION 3.20)
project(SecureLinkOpenVPN3Evaluation LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DEFINED OPENVPN3_SOURCE OR NOT DEFINED ASIO_SOURCE)
  message(FATAL_ERROR "Use scripts/build-openvpn3-eval with pinned source dependencies")
endif()
list(PREPEND CMAKE_MODULE_PATH "${OPENVPN3_SOURCE}/cmake")
set(ASIO_INCLUDE_DIR "${ASIO_SOURCE}/asio/include" CACHE PATH "Pinned standalone Asio" FORCE)
include(findcoredeps)
add_executable(szu-openvpn3-eval main.cpp "${OPENVPN3_SOURCE}/client/ovpncli.cpp")
add_core_dependencies(szu-openvpn3-eval)
target_compile_definitions(szu-openvpn3-eval PRIVATE USE_TUN_BUILDER)
target_compile_options(szu-openvpn3-eval PRIVATE -include "${CMAKE_CURRENT_SOURCE_DIR}/quiet.hpp")
# Homebrew's common include directory may also contain OpenSSL 4 headers.
# Ensure the selected pkg-config version wins over unrelated dependencies.
pkg_get_variable(EVAL_OPENSSL_INCLUDEDIR openssl includedir)
target_include_directories(szu-openvpn3-eval SYSTEM BEFORE PRIVATE "${EVAL_OPENSSL_INCLUDEDIR}")
if(NOT OpenSSL_VERSION MATCHES "^3\\.")
  message(FATAL_ERROR "Evaluator requires selected OpenSSL 3; use a fresh out/ build directory after changing dependencies")
endif()

# Separate, explicitly enabled handshake probe; no change to the eval-only executable.
if(APPLE)
  enable_language(OBJCXX)
  set(CMAKE_OBJCXX_STANDARD 17)
  set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
  add_executable(szu-openvpn3-handshake ../openvpn3-handshake/main.mm "${OPENVPN3_SOURCE}/client/ovpncli.cpp")
  add_core_dependencies(szu-openvpn3-handshake)
  if(NOT DEFINED OPENVPN3_OVERLAY)
    message(FATAL_ERROR "Use scripts/build-openvpn3-eval for the reviewed TCP transport overlay")
  endif()
  target_compile_definitions(szu-openvpn3-handshake PRIVATE USE_TUN_BUILDER OPENVPN_EXTERNAL_TRANSPORT_FACTORY)
  target_include_directories(szu-openvpn3-handshake BEFORE PRIVATE "${OPENVPN3_OVERLAY}")
  target_compile_options(szu-openvpn3-handshake PRIVATE -include "${CMAKE_CURRENT_SOURCE_DIR}/quiet.hpp")
  target_include_directories(szu-openvpn3-handshake SYSTEM BEFORE PRIVATE "${EVAL_OPENSSL_INCLUDEDIR}")
  target_link_libraries(szu-openvpn3-handshake "-framework Foundation")

  # Merged data-session engine: OpenVPN3 session plus the audited executor
  # transaction client in one binary. Built from the same pinned sources and
  # overlays; the privileged executor stages its cdhash out of band.
  add_executable(szu-openvpn3-data-engine ../openvpn3-handshake/data_engine.mm ../network-helper/ipc.mm "${OPENVPN3_SOURCE}/client/ovpncli.cpp")
  add_core_dependencies(szu-openvpn3-data-engine)
  target_compile_definitions(szu-openvpn3-data-engine PRIVATE USE_TUN_BUILDER OPENVPN_EXTERNAL_TRANSPORT_FACTORY)
  target_include_directories(szu-openvpn3-data-engine BEFORE PRIVATE "${OPENVPN3_OVERLAY}")
  target_include_directories(szu-openvpn3-data-engine PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../network-helper")
  target_compile_options(szu-openvpn3-data-engine PRIVATE -include "${CMAKE_CURRENT_SOURCE_DIR}/quiet.hpp")
  target_include_directories(szu-openvpn3-data-engine SYSTEM BEFORE PRIVATE "${EVAL_OPENSSL_INCLUDEDIR}")
  target_link_libraries(szu-openvpn3-data-engine "-framework Foundation" "-framework Security" "-lbsm")
endif()
